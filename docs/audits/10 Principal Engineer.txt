Refactor 2: Non-Destructive Opposing Touch Arbitration (KiepieBike.gd)

    Target Files: KiepieBike.gd

    Why It Matters: Prevents opposing multi-touch vectors from canceling throttle or oscillating reverse state during high-speed turns.

    Blueprint:

        Before (Conceptual):

GDScript

# Naive additive or overwrite touch accumulation
var throttle_input: float = touch_forward_strength - touch_reverse_strength
if abs(throttle_input) < 0.05:
    engine_force = 0.0

    After (Conceptual):

GDScript

const INPUT_DEADZONE: float = 0.12

func _resolve_drive_input(kbd_axis: float, touch_fwd: float, touch_rev: float, forward_speed: float) -> float:
    var active_fwd: float = touch_fwd if touch_fwd > INPUT_DEADZONE else 0.0
    var active_rev: float = touch_rev if touch_rev > INPUT_DEADZONE else 0.0
    
    # Explicit conflict resolution: active braking/reverse overrides throttle when both touches are held
    var touch_axis: float = 0.0
    if active_rev > 0.0 and active_fwd > 0.0:
        touch_axis = -active_rev if forward_speed > 1.5 else active_fwd
    else:
        touch_axis = active_fwd - active_rev
        
    return clampf(kbd_axis if absf(kbd_axis) > absf(touch_axis) else touch_axis, -1.0, 1.0)
